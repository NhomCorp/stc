export NODE_OPTIONS=--max-old-space-size=2048
export NEXT_TELEMETRY_DISABLED=1
npm ci   # chỉ khi cần
npm run build

CTX=/tmp/stc-image-context
rm -rf "$CTX" && mkdir -p "$CTX/.next/static"
cp -a public "$CTX/public"
cp -a .next/standalone/. "$CTX/"
cp -a .next/static/. "$CTX/.next/static/"
cat > "$CTX/Dockerfile" <<'EOF'
FROM node:22-alpine
RUN addgroup -S -g 1001 nextjs && adduser -S -u 1001 nextjs
WORKDIR /app
ENV NODE_ENV=production NEXT_TELEMETRY_DISABLED=1 PORT=3000 HOSTNAME=0.0.0.0
COPY --chown=nextjs:nextjs . .
USER nextjs
EXPOSE 3000
CMD ["node", "server.js"]
EOF
docker build -t stc-app:latest "$CTX"

# 4) Recreate container (network = network của stc-postgres)
NET=$(docker inspect stc-postgres -f '{{range $k,$v := .NetworkSettings.Networks}}{{println $k}}{{end}}' | head -1)

set -a; . ./.env; set +a
docker rm -f stc-app
docker run -d --name stc-app --restart unless-stopped \

  -e DATABASE_URL="$DATABASE_URL" -e PORT=3000 \

<script>
  function copyCode(btn) {
    const code = btn.closest('.code-wrapper').querySelector('pre code').innerText;
    navigator.clipboard.writeText(code).then(() => {
      const original = btn.innerHTML;
      btn.classList.add('copied');
      btn.innerHTML = 'Đã copy!';
      const toast = document.getElementById('toast');
      toast.classList.add('show');
      setTimeout(() => {
        btn.classList.remove('copied');
        btn.innerHTML = original;
        toast.classList.remove('show');
      }, 1800);
    });
  }
